cd /data1/jhc/storage_of_code/limap_note
conda activate limap

CUDA_VISIBLE_DEVICES="$device" \
PYTORCH_ALLOC_CONF=expandable_segments:True \
python runners/colmap_triangulation.py \
--colmap_path "$S" \
--model_path sparse/0 \
--image_path images \
--output_dir "$S"/triangulation \
-nv 4 \
--line2d.max_num_2d_segs 15000 \
--line2d.detector.method opencv_lsd \
--max_image_dim 1600 \
--skip_exists

colmap model_converter \
--input_path "$S"/sparse/0 \
--output_path "$S"/sparse/0 \
--output_type BIN

cd /data1/jhc/storage_of_code/Depth-Anything-V2
conda activate depth-anything

CUDA_DEVICE_ORDER=PCI_BUS_ID CUDA_VISIBLE_DEVICES="$device" \
python run.py \
--encoder vitl \
--img-path "$S"/images \
--outdir "$S"/depth \
--input-size 518 \
--pred-only \
--grayscale \
--npy

cd /data1/jhc/storage_of_code/IndoorGS-note/geomtric-cues
conda activate indoorGS

CUDA_DEVICE_ORDER=PCI_BUS_ID \
CUDA_VISIBLE_DEVICES="$device" \
python extract_plane_cues.py \
--data_root "$S" \
--sam_checkpoint /data1/jhc/storage_of_code/IndoorGS-note/geomtric-cues/sam_model/sam_vit_h_4b8939.pth \
--line_dir "$S"/triangulation \
--output_path "$S"/sparse/0/points3D_plane.bin \
--max_resolution 1024

python sor_points3D.py --input "$S"

python extract_line_points.py --input "$S"

cd ..

CUDA_DEVICE_ORDER=PCI_BUS_ID \
CUDA_VISIBLE_DEVICES="$device" \
TORCH_HOME=/data1/jhc/torch_cache \
python -u train.py \
--source_path "$S" \
--model_path "$M" \
--disable_viewer \
--eval

CUDA_DEVICE_ORDER=PCI_BUS_ID \
CUDA_VISIBLE_DEVICES="$device" \
TORCH_HOME=/data1/jhc/torch_cache \
python render.py \
--source_path "$S" \
--model_path "$M" \
--iteration 30000 \
--skip_train

CUDA_DEVICE_ORDER=PCI_BUS_ID \
CUDA_VISIBLE_DEVICES="$device" \
TORCH_HOME=/data1/jhc/torch_cache \
python metrics.py \
--model_paths "$M" \
--iteration 30000 \
--skip_train
